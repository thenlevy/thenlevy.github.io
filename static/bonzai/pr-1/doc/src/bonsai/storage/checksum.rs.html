19fn mix(sum: u32, value: u32) -> u32 {
20    let tmp = sum ^ value;
21    tmp.wrapping_mul(FNV_PRIME) ^ (tmp >> 17)
22}

24fn checksum_block(page: &Page) -> u32 {
25    let mut sums = BASE_OFFSETS;
26    for (i, row) in page.chunks_exact(4 * N_SUMS).enumerate() {
27        for (j, (sum, word)) in sums.iter_mut().zip(row.chunks_exact(4)).enumerate() {
28            let mut value = u32::from_le_bytes(word.try_into().unwrap());
29            // Bytes 8..12: pd_checksum in the low half, pd_flags in the high
30            // half. The page's own checksum must read as zero.
31            if i == 0 && j == 2 {
32                value &= 0xFFFF_0000;
33            }
34            *sum = mix(*sum, value);
35        }
36    }
37    // Two rounds of zeros mix the last words in as well as the others.
38    for _ in 0..2 {
39        for sum in &mut sums {
40            *sum = mix(*sum, 0);
41        }
42    }
43    sums.iter().fold(0, |result, sum| result ^ sum)
44}

46/// The checksum `page` should carry when stored at `block`. The page's own
47/// `pd_checksum` field is read as zero.
48pub fn checksum_page(page: &Page, block: BlockNumber) -> u16 {
49    // Mixing in the block number catches a valid page written to the wrong place.
50    let checksum = checksum_block(page) ^ block.0;
51    // Fold to 16 bits, plus one so that a checksum is never zero.
52    (checksum % 65535 + 1) as u16
53}
